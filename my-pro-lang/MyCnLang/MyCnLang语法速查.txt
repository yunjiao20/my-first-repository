    # 函数定义
    函数 相加（a，b）（
        返回 a + b
    ）

    def 相加(a, b)(
        return a + b
    ) # 与上面的等价

    def say_hi()( print('hi') )

    # 函数调用
    相加（1，2）  # 函数式
    相加  1  2    # 命令式，返回值存放在@r，就如bash的$?
    call say_hi   # 无参调用，对于需要参数的函数不要使用这个方法
